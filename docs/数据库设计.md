# 数据库设计

**版本**：v1.0  
**日期**：2026-04-17  
**数据库**：SQLite（better-sqlite3）  
**ORM**：Drizzle ORM

---

## 1. 设计原则

- **本地优先**：所有数据存储于用户设备，不上传任何数据
- **WAL 模式**：启用 Write-Ahead Logging，提高并发读写性能
- **轻量**：避免过度设计，表结构以实用为导向
- **可迁移**：使用 Drizzle 迁移系统，每次 schema 变更通过迁移文件执行

---

## 2. 数据库配置

```typescript
// database/DatabaseService.ts
import Database from 'better-sqlite3';
import { drizzle } from 'drizzle-orm/better-sqlite3';
import { migrate } from 'drizzle-orm/better-sqlite3/migrator';
import path from 'path';
import { app } from 'electron';

export function createDatabase() {
  const dbPath = path.join(app.getPath('userData'), 'cubicle.db');
  const sqlite = new Database(dbPath);

  // 性能与可靠性配置
  sqlite.pragma('journal_mode = WAL');       // WAL 模式
  sqlite.pragma('synchronous = NORMAL');     // 性能与安全的平衡
  sqlite.pragma('cache_size = -64000');      // 64MB 缓存
  sqlite.pragma('foreign_keys = ON');        // 启用外键约束
  sqlite.pragma('temp_store = MEMORY');      // 临时表存内存

  const db = drizzle(sqlite);

  // 运行迁移
  migrate(db, { migrationsFolder: path.join(__dirname, 'migrations') });

  return db;
}
```

## 3. 表结构设计

### 3.1 用户表（users）

存储本机用户信息以及已知的局域网用户缓存。

```typescript
// database/schema.ts
import { sqliteTable, text, integer, blob } from 'drizzle-orm/sqlite-core';

export const users = sqliteTable('users', {
  // 主键
  userId: text('user_id').primaryKey(),        // UUID，全局唯一

  // 基本信息
  nickname: text('nickname').notNull(),
  avatarData: blob('avatar_data'),              // 64x64 JPEG binary，可为空
  status: text('status')
    .notNull()
    .default('online'),                         // 'online'|'busy'|'away'|'offline'

  // 网络信息
  ipAddress: text('ip_address'),                // 最后已知 IP
  udpPort: integer('udp_port').default(2425),
  tcpPort: integer('tcp_port').default(2426),
  clientVersion: text('client_version'),

  // 元数据
  isSelf: integer('is_self', { mode: 'boolean' }).notNull().default(false),
  isBlocked: integer('is_blocked', { mode: 'boolean' }).notNull().default(false),
  groupName: text('group_name'),                // 用户分组（本地设置）
  remark: text('remark'),                       // 备注名（本地设置）
  lastSeenAt: integer('last_seen_at'),          // Unix timestamp（毫秒）
  firstSeenAt: integer('first_seen_at').notNull(),
  updatedAt: integer('updated_at').notNull(),
});
```

**索引：**
```sql
CREATE INDEX idx_users_ip ON users(ip_address);
CREATE INDEX idx_users_status ON users(status);
```

### 3.3 消息表（messages）

所有消息的持久化存储。

```typescript
export const messages = sqliteTable('messages', {
  messageId: text('message_id').primaryKey(),    // UUID（与 UDP msgId 一致）

  // 关联
  conversationId: text('conversation_id')
    .notNull()
    .references(() => conversations.conversationId, { onDelete: 'cascade' }),
  senderId: text('sender_id').notNull(),          // 发送方 userId

  // 消息内容
  contentType: text('content_type').notNull(),
  // 'text' | 'emoji' | 'image' | 'file' | 'system' | 'recall'
  content: text('content'),                       // 文本内容（contentType='text/emoji'）
  fileId: text('file_id'),                        // 关联文件（contentType='image/file'）

  // 引用回复
  replyToId: text('reply_to_id'),                 // 被引用消息的 messageId

  // 状态
  status: text('status').notNull().default('sent'),
  // 'sending' | 'sent' | 'delivered' | 'failed' | 'recalled'
  isRecalled: integer('is_recalled', { mode: 'boolean' }).notNull().default(false),

  // 时间
  sentAt: integer('sent_at').notNull(),            // 发送时间（毫秒）
  deliveredAt: integer('delivered_at'),            // 送达时间
  createdAt: integer('created_at').notNull(),      // 本地写入时间
});
```

**索引：**
```sql
CREATE INDEX idx_messages_conversation ON messages(conversation_id, sent_at DESC);
CREATE INDEX idx_messages_sender ON messages(sender_id);
CREATE INDEX idx_messages_file ON messages(file_id);
-- 全文搜索（SQLite FTS5）
CREATE VIRTUAL TABLE messages_fts USING fts5(
  message_id UNINDEXED,
  content,
  content='messages',
  content_rowid='rowid'
);
```

## 4. 数据访问层（Repository）

### 4.1 MessageRepository

```typescript
// repositories/MessageRepo.ts
export class MessageRepository {
  constructor(private db: DrizzleDB) {}

  async save(message: NewMessage): Promise<Message> {
    const [saved] = await this.db
      .insert(messages)
      .values(message)
      .returning();
    return saved;
  }

  async findByConversation(
    conversationId: string,
    options: { limit: number; before?: number }
  ): Promise<Message[]> {
    return this.db
      .select()
      .from(messages)
      .where(
        and(
          eq(messages.conversationId, conversationId),
          options.before
            ? lt(messages.sentAt, options.before)
            : undefined
        )
      )
      .orderBy(desc(messages.sentAt))
      .limit(options.limit);
  }

  async search(keyword: string, conversationId?: string): Promise<Message[]> {
    // 使用 FTS5 全文搜索
    const query = conversationId
      ? `SELECT * FROM messages_fts WHERE content MATCH ? AND message_id IN (
           SELECT message_id FROM messages WHERE conversation_id = ?
         ) ORDER BY rank LIMIT 50`
      : `SELECT * FROM messages_fts WHERE content MATCH ? ORDER BY rank LIMIT 50`;

    return this.db.run(sql.raw(query, [keyword, conversationId].filter(Boolean)));
  }



  async recall(messageId: string): Promise<void> {
    await this.db
      .update(messages)
      .set({ isRecalled: true, status: 'recalled', content: null })
      .where(eq(messages.messageId, messageId));
  }
}
```

### 4.2 UserRepository

```typescript
export class UserRepository {
  async upsert(user: NewUser): Promise<User> {
    const [saved] = await this.db
      .insert(users)
      .values(user)
      .onConflictDoUpdate({
        target: users.userId,
        set: {
          nickname: user.nickname,
          ipAddress: user.ipAddress,
          status: user.status,
          lastSeenAt: Date.now(),
          updatedAt: Date.now(),
        },
      })
      .returning();
    return saved;
  }

  async setOffline(userId: string): Promise<void> {
    await this.db
      .update(users)
      .set({ status: 'offline', updatedAt: Date.now() })
      .where(eq(users.userId, userId));
  }

  async findOnline(): Promise<User[]> {
    return this.db
      .select()
      .from(users)
      .where(
        and(
          eq(users.isSelf, false),
          eq(users.isBlocked, false),
          ne(users.status, 'offline')
        )
      )
      .orderBy(asc(users.nickname));
  }
}
```

## 5. 数据清理策略

### 5.1 定时清理

每天凌晨 3:00（或应用启动时检查上次清理时间）执行：

```typescript
async function cleanupOldData(retentionDays: number) {
  const cutoff = Date.now() - retentionDays * 24 * 60 * 60 * 1000;

  // 删除旧消息（保留文件记录）
  await db.delete(messages)
    .where(lt(messages.sentAt, cutoff));

  // 删除已完成的旧文件传输记录（实际文件不删除）
  await db.delete(files)
    .where(
      and(
        lt(files.completedAt, cutoff),
        eq(files.status, 'completed')
      )
    );

  // 清理孤立会话（无消息的非群组会话）
  // ...

  // 压缩数据库
  db.exec('VACUUM');
}
```

### 5.2 数据库大小估算

| 场景 | 条目数/天 | 单条大小 | 180天占用 |
|------|---------|---------|---------|
| 文本消息（活跃用户） | 500条 | ~300 B | ~27 MB |
| 文件记录（元数据） | 20条 | ~500 B | ~1.8 MB |
| 用户缓存 | 50人 | ~1 KB | ~50 KB |
| **合计** | - | - | **< 30 MB** |

## 6. 数据库迁移

### 6.1 初始迁移（0000_init.sql）

```sql
-- 由 Drizzle 根据 schema.ts 自动生成
CREATE TABLE users (...);
CREATE TABLE conversations (...);
CREATE TABLE messages (...);
CREATE TABLE files (...);
CREATE TABLE settings (...);

-- FTS5 虚拟表
CREATE VIRTUAL TABLE messages_fts USING fts5(
  message_id UNINDEXED,
  content,
  content='messages',
  content_rowid='rowid'
);

-- 触发器：同步 FTS 索引
CREATE TRIGGER messages_fts_insert AFTER INSERT ON messages BEGIN
  INSERT INTO messages_fts(rowid, message_id, content)
  VALUES (new.rowid, new.message_id, new.content);
END;

CREATE TRIGGER messages_fts_delete AFTER DELETE ON messages BEGIN
  INSERT INTO messages_fts(messages_fts, rowid, message_id, content)
  VALUES('delete', old.rowid, old.message_id, old.content);
END;
```
